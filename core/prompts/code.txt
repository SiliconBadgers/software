def merge_intervals(intervals):
    """Merge overlapping closed intervals.

    Takes a list of (start, end) pairs and returns a new list of
    non-overlapping pairs sorted by start. Touching intervals such as
    (1, 3) and (3, 5) are merged into (1, 5).
    """
    if not intervals:
        return []
    ordered = sorted(intervals, key=lambda pair: pair[0])
    merged = [ordered[0]]
    for start, end in ordered[1:]:
        last_start, last_end = merged[-1]
        if start <= last_end:
            merged[-1] = (last_start, max(last_end, end))
        else:
            merged.append((start, end))
    return merged


def test_merge_intervals():
    assert merge_intervals([]) == []
    assert merge_intervals([(1, 3), (2, 6), (8, 10)]) == [(1, 6), (8, 10)]
    assert merge_intervals([(1, 3), (3, 5)]) == [(1, 5)]
    assert merge_intervals([(5, 7), (1, 2)]) == [(1, 2), (5, 7)]


if __name__ == "__main__":
    test_merge_intervals()
    print("all tests passed")
